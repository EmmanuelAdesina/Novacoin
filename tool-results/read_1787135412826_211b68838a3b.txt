    24→/* ═══════════════════════════════════════════════════════════
    25→   ANIMATED NUMBER
    26→   ═══════════════════════════════════════════════════════════ */
    27→function AnimNum({ value, className = '' }: { value: number; className?: string }) {
    28→  const ref = useRef<HTMLSpanElement>(null);
    29→  const [display, setDisplay] = useState(0);
    30→  const prev = useRef(0);
    31→  useEffect(() => {
    32→    const start = prev.current;
    33→    const end = value;
    34→    prev.current = end;
    35→    if (start === end) return;
    36→    const dur = 600;
    37→    const t0 = performance.now();
    38→    function tick(now: number) {
    39→      const p = Math.min((now - t0) / dur, 1);
    40→      const e = 1 - Math.pow(1 - p, 3);
    41→      setDisplay(Math.round(start + (end - start) * e));
    42→      if (p < 1) requestAnimationFrame(tick);
    43→    }
    44→    requestAnimationFrame(tick);
    45→  }, [value]);
    46→  return <span ref={ref} className={className}>{display.toLocaleString()}</span>;
    47→}

    52→function Reveal({ children, delay = 0, className = '', direction = 'up' }: {
    53→  children: React.ReactNode; delay?: number; className?: string; direction?: 'up' | 'left' | 'right' | 'scale';
    54→}) {
    55→  const ref = useRef(null);
    56→  const inView = useInView(ref, { once: true, margin: '-80px' });
    57→  const initial = direction === 'left' ? { opacity: 0, x: -40 } : direction === 'right' ? { opacity: 0, x: 40 } : direction === 'scale' ? { opacity: 0, scale: 0.85 } : { opacity: 0, y: 30 };
    58→  return (
    59→    <motion.div ref={ref} initial={initial} animate={inView ? { opacity: 1, x: 0, y: 0, scale: 1 } : {}} transition={{ duration: 0.65, delay, ease }} className={className}>
    60→      {children}
    61→    </motion.div>
    62→  );
    63→}

    85→function SoftButton({ children, onClick, variant = 'primary', className = '' }: {

    87→}) {

    93→  return (
    94→    <motion.button onClick={onClick} onMouseEnter={() => setHovered(true)} onMouseLeave={() => setHovered(false)} whileTap={{ scale: 0.96 }} className={`${base} ${styles} ${className}`}>
    95→      {children}
